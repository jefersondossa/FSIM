490 void Element<2>::getJacobianMatrix(ublas::bounded_vector<double,2>& xsi) {
491 
492  //Computes the spatial Jacobian matrix and its inverse
493 
494  typename QuadShapeFunction<2>::ValueDeriv dphi;
495 
496  shapeQuad.evaluateGradient(xsi,dphi);
497 
498  double dx_dxsi1 = 0.;
499  double dx_dxsi2 = 0.;
500  double dy_dxsi1 = 0.;
501  double dy_dxsi2 = 0.;
502 
503  for (int i=0; i<6; i++){
504  dx_dxsi1 += localNodes_(i,0) * dphi(0,i);
505  dx_dxsi2 += localNodes_(i,0) * dphi(1,i);
506  dy_dxsi1 += localNodes_(i,1) * dphi(0,i);
507  dy_dxsi2 += localNodes_(i,1) * dphi(1,i);
508  };
509 
510  //Computing the jacobian determinant
511  djac_ = dx_dxsi1 * dy_dxsi2 - dx_dxsi2 * dy_dxsi1;
512  // djac_ = fabs(djac_);
513 
514  //Computing Jacobian inverse
515  ainv_(0,0) = dy_dxsi2 / djac_;
516  ainv_(0,1) = -dy_dxsi1 / djac_;
517  ainv_(1,0) = -dx_dxsi2 / djac_;
518  ainv_(1,1) = dx_dxsi1 / djac_;
519 
520  return;
