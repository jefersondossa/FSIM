#include "OpenGL_WC.h"

void Playback::InitializeGL() {
    glClearColor(1.0f, 1.0f, 1.0f, 1); // BackGround Color
    glEnable(GL_DEPTH_TEST); // Enable depth testing. Helps determine which objects are in front of others when they are drawn on the screen.
}

void Playback::draw() {
    static bool firstTime = true;
    if (firstTime) {
        InitializeGL();
        firstTime = false;
    }

    glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT); // Clears the color and depth buffer

    // Get current window size
    int w = this->w();
    int h = this->h();

    // Adjust the viewport
    glViewport(0, 0, w, h);

    // view transformations
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    float aspect = static_cast<float>(w) / static_cast<float>(h);
    gluPerspective(45.0, aspect, 1.0, 100.0); // Perspective projection with aspect ratio

    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();
    gluLookAt(0, 0, 3 + zoom, 0, 0, 0, 0, 1, 0); // Apply zoom here

    // Apply rotations
    glRotatef(rotationX, 1, 0, 0);
    glRotatef(rotationY, 0, 1, 0);

    // Draw the cube
    DrawCube();

    // Draw the axes
    DrawAxes();
}

void Playback::DrawCube() {
    // Specify vertices of the cube
    GLfloat vertices[] = {
        // Front face
        -1, 1, 1,    // Vertex 0
        -1, -1, 1,   // Vertex 1
        1, -1, 1,    // Vertex 2
        1, 1, 1,     // Vertex 3

        // Back face
        -1, 1, -1,   // Vertex 4
        -1, -1, -1,  // Vertex 5
        1, -1, -1,   // Vertex 6
        1, 1, -1     // Vertex 7
    };

    // Indices of vertices for each line segment
    GLuint indices[] = {
        // Front face
        0, 1,
        1, 2,
        2, 3,
        3, 0,

        // Back face
        4, 5,
        5, 6,
        6, 7,
        7, 4,

        // Connecting lines
        0, 4,
        1, 5,
        2, 6,
        3, 7
    };

    // Set line color to white
    glColor3f(0.0f, 0.0f, 0.0f);

    // Set line thickness
    glLineWidth(2.0f);

    // Render the cube
    glBegin(GL_LINES);
    for (int i = 0; i < sizeof(indices) / sizeof(indices[0]); i += 2) {
        glVertex3fv(&vertices[3 * indices[i]]);
        glVertex3fv(&vertices[3 * indices[i + 1]]);
    }
    glEnd();
}

void Playback::DrawAxes() {
     // Set viewport for the axes in the corner
    glViewport(10, 10, 100, 100);

    // Set up orthogonal projection
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    glOrtho(-1, 1, -1, 1, -1, 1);

    // Set modelview matrix
    glMatrixMode(GL_MODELVIEW);
    glLoadIdentity();

    // Apply the same rotations as the main view
    glRotatef(rotationX, 1, 0, 0);
    glRotatef(rotationY, 0, 1, 0);

    // Set line width for the axes within the supported range
    glLineWidth(2.0f); // Adjust this value within the supported range

    // Draw the axes
    glBegin(GL_LINES);
    // X axis in red
    glColor3f(0, 0, 0);
    glVertex3f(0, 0, 0);
    glVertex3f(0.5, 0, 0);

    // Y axis in green
    glColor3f(0, 0, 0);

    // Z axis in blue
    glColor3f(0, 0, 0);

    glVertex3f(0, 0, 0.5);

    glRasterPos3f(0, 0.55, 0);

    glRasterPos3f(0, 0, 0.55);
    glColor3f(0, 0, 0);
    glutBitmapCharacter(GLUT_BITMAP_HELVETICA_18, 'Z');
}

int Playback::handle(int event) {
    switch (event) {
    case FL_PUSH:
        last_x = Fl::event_x();
        last_y = Fl::event_y();
        rotating = true;
        return 1;

    case FL_DRAG:
        if (rotating) {
            int dx = Fl::event_x() - last_x;
            int dy = Fl::event_y() - last_y;

            rotationX += dy; // Update rotation angle for X axis
            rotationY += dx; // Update rotation angle for Y axis

            last_x = Fl::event_x();
            last_y = Fl::event_y();

            redraw();
            return 1;
        }
        break;

    case FL_RELEASE:
        rotating = false;
        return 1;

    case FL_MOUSEWHEEL:
        zoom += Fl::event_dy() * 0.1f; // Adjust zoom based on scroll direction
        redraw();
        return 1;

    default:
        break;
    }

    return Fl_Gl_Window::handle(event);
}

void IdleCallback(void* pData) {
    if (pData != NULL) {
        Playback* pWindow = reinterpret_cast<Playback*>(pData);
        if (pWindow->animating) {
            pWindow->rotation += pWindow->rotationIncrement / 100;
            pWindow->redraw();
        }
    }
}

Playback::Playback(int X, int Y, int W, int H, const char *L) 
    : Fl_Gl_Window(X, Y, W, H, L), last_x(0), last_y(0), rotating(false), 
      rotationX(0), rotationY(0), zoom(3) { // Initialize zoom to -1.0 to start further back
    mode(FL_RGB | FL_ALPHA | FL_DEPTH | FL_DOUBLE);
    rotation = 0.f;
    rotationIncrement = 10.f;
    animating = false;
    Fl::add_idle(IdleCallback, this);
    end();
}

void Playback::ReadGeometry(const string filename){

    // Here, filename is that same as gp at Implem_C1_WC.cpp

    fstream file(filename);
    string line;
    file.open(filename, ios::in);

    if(file.is_open()){
        while(getline(file,line)){

            //Verifys if the line beggins with "//"
            if (line.find("//") == 0) {

            //Ignores the line and follows to the next interation loop              
            continue;
            }

            //Declares the "match" name for each pattern found in the "filename" file:
            smatch match;

            //Creates a pattern that reads "Point(i) = {x, y, z, m};", where:
            //  match[1] = point id;
            //  match[2] = x;
            //  match[3] = y;
            //  match[4] = z;
            regex point_regex(R"(Point\s*\((\d+)\)\s*=\s*\{([-+]?\d*\.\d+|\d+),([-+]?\d*\.\d+|\d+),([-+]?\d*\.\d+|\d+),\w+\})");
            //Creates a pattern that reads "Line(j) = {a, b};" | {a, b} E i, where:
            //  match[1] = line id;
            //  match[2] = point a;
            //  match[3] = point b;
            regex line_regex(R"(Line\s*\((\d+)\)\s*=\s*\{(\d+),(\d+)\})");

            //Searchs in every line a match to the point_pattern 
            if (std::regex_search(line, match, point_regex)){
                
                int point_id = stof(match[1]);
                float x = stof(match[2]);
                float y = stof(match[3]);
                float z = stof(match[4]);

                points.emplace_back(point_id, x, y, z); 

            }

            else if (std::regex_search(line, match, line_regex)){
                
                int line_id = stof(match[1]);
                float a = stof(match[2]);
                float b = stof(match[3]);

                lines.emplace_back(a - 1, b - 1);// Store 0-based indices points for lines, since the vertexs begins in 0.
            }
        }
    }

    file.close();
}
